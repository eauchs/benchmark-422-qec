    ops\[q\] \= op

def embed\_2q\_op\_fast(op2q, q1, q2, n=N):  
    """Embed 2-qubit op on qubits q1, q2"""  
    \# Use tensor product decomposition via Pauli expansion  
    dim \= 2\*\*n  
    result \= np.zeros((dim, dim), dtype=complex)  
    for i in range(dim):  
        for j in range(dim):  
            def gb(x, q): return (x \>\> (n \- 1 \- q)) & 1  
            i1, i2 \= gb(i,q1), gb(i,q2)  
            j1, j2 \= gb(j,q1), gb(j,q2)  
            if all(gb(i,q)==gb(j,q) for q in range(n) if q\!=q1 and q\!=q2):  
                result\[i,j\] \= op2q\[2\*i1+i2, 2\*j1+j2\]  
    return result

\# r\_mu \-\> lambda \* sum\_nu \[CNOT action on Paulis\] r\_nu  
\# where only the Pauli components on the 2 affected qubits change.  
\# Non-identity components get multiplied by lambda.  
\# Identity stays unchanged.  
\#

\# and E(I) \= I

    dim \= 2\*\*n  
    result \= np.zeros((dim, dim), dtype=complex)  
    for i in range(dim):  
        for j in range(dim):  
            def gb(x, q): return (x \>\> (n \- 1 \- q)) & 1  
            i1, i2 \= gb(i,q1), gb(i,q2)
